    <script>
        let score = 0;
        let pps = 0;
        let clickValue = 1;
        let autoCost = 15;
        let multiCost = 50;
        function clickEff(e) {
            score += clickValue;
            updateUI();
            
            // Sayı uçurma efekti
            const float = document.createElement('div');
            float.className = 'floating-text';
            float.innerText = '+' + clickValue;
            float.style.left = e.clientX + 'px';
            float.style.top = e.clientY + 'px';
            document.body.appendChild(float);
            setTimeout(() => float.remove(), 800);
        }
        function buyUpgrade(type) {
            if (type === 'auto' && score >= autoCost) {
                score -= autoCost;
                pps += 1;
                autoCost = Math.round(autoCost * 1.5);
                updateUI();
            } else if (type === 'multi' && score >= multiCost) {
                score -= multiCost;
                clickValue += 2;
                multiCost = Math.round(multiCost * 2);
                updateUI();
            }
        }
        function updateUI() {
            document.getElementById('score').innerText = Math.floor(score);
            document.getElementById('pps').innerText = "Saniyelik Kazanç: " + pps;
            document.getElementById('autoBtn').innerText = autoCost + " Puan";
            document.getElementById('multiBtn').innerText = multiCost + " Puan";
        }
        // Otomatik puan sistemi
        setInterval(() => {
            score += pps;
            updateUI();
        }, 1000);
    </script>
